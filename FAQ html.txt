  <script>
    // 첨부 JSON에 있는 FAQ 데이터만 사용
    const faqData = [
      {
        category: "교육",
        question: "교육 신청은 어디에서 하나요?",
        answer_seed: "교육 포털의 과정 검색 후 신청 메뉴를 사용합니다."
      },
      {
        category: "교육",
        question: "신청한 교육을 취소하려면 어떻게 하나요?",
        answer_seed: "신청내역에서 취소 가능 여부를 확인합니다."
      },
      {
        category: "문서",
        question: "주간 업무보고에는 무엇을 넣어야 하나요?",
        answer_seed:
          "주요 실적, 이슈, 다음 주 계획을 중심으로 작성합니다."
      },
      {
        category: "IT",
        question: "사내 시스템 접속이 되지 않습니다.",
        answer_seed:
          "네트워크, 브라우저, 계정 상태를 순서대로 확인합니다."
      },
      {
        category: "보안",
        question: "ChatGPT에 내부 업무자료를 넣어도 되나요?",
        answer_seed:
          "교육에서는 합성자료를 사용하고, 실제 자료는 조직의 보안 및 AI 사용 기준을 먼저 확인합니다."
      },
      {
        category: "복지-카페",
        question: "사내 카페시설을 이용하려면 어떻게 해야하나요?",
        answer_seed:
          "부여받은 포인트로 음료 구매가 가능하며, 포인트 사용은 사내카페의 키오스크에 사원증 태그를 통해 이루어집니다 ."
      }

    ];

    const searchInput = document.getElementById("searchInput");
    const categoryFilters = document.getElementById("categoryFilters");
    const faqList = document.getElementById("faqList");
    const resultCount = document.getElementById("resultCount");
    const resetBtn = document.getElementById("resetBtn");

    let selectedCategory = "전체";

    function escapeHtml(text) {
      return String(text)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    function escapeRegExp(text) {
      return text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    }

    function highlight(text, keyword) {
      const safeText = escapeHtml(text);
      const trimmedKeyword = keyword.trim();

      if (!trimmedKeyword) {
        return safeText;
      }

      const regex = new RegExp(
        `(${escapeRegExp(trimmedKeyword)})`,
        "gi"
      );

      return safeText.replace(regex, "<mark>$1</mark>");
    }

    function getFilteredFaqs() {
      const keyword = searchInput.value.trim().toLowerCase();

      return faqData.filter((item) => {
        const matchesCategory =
          selectedCategory === "전체" ||
          item.category === selectedCategory;

        const matchesKeyword =
          !keyword ||
          item.question.toLowerCase().includes(keyword) ||
          item.answer_seed.toLowerCase().includes(keyword);

        return matchesCategory && matchesKeyword;
      });
    }

    function renderFaqs() {
      const keyword = searchInput.value.trim();
      const filteredFaqs = getFilteredFaqs();

      resultCount.textContent = filteredFaqs.length;

      if (filteredFaqs.length === 0) {
        faqList.innerHTML = `
          <div class="empty-state">
            조건에 맞는 FAQ가 없습니다.<br>
            다른 키워드나 분류를 선택해 주세요.
          </div>
        `;
        return;
      }

      faqList.innerHTML = filteredFaqs
        .map(
          (item) => `
            <article class="faq-card">
              <span class="category-badge">
                ${escapeHtml(item.category)}
              </span>

              <h3 class="question">
                ${highlight(item.question, keyword)}
              </h3>

              <p class="answer">
                ${highlight(item.answer_seed, keyword)}
              </p>
            </article>
          `
        )
        .join("");
    }

    // 키워드 검색
    searchInput.addEventListener("input", renderFaqs);

    // 카테고리 필터
    categoryFilters.addEventListener("click", (event) => {
      const button = event.target.closest(".filter-btn");

      if (!button) return;

      selectedCategory = button.dataset.category;

      document.querySelectorAll(".filter-btn").forEach((btn) => {
        btn.classList.toggle("active", btn === button);
      });

      renderFaqs();
    });

    // 전체보기
    resetBtn.addEventListener("click", () => {
      searchInput.value = "";
      selectedCategory = "전체";

      document.querySelectorAll(".filter-btn").forEach((btn) => {
        btn.classList.toggle(
          "active",
          btn.dataset.category === "전체"
        );
      });

      renderFaqs();
      searchInput.focus();
    });

    // 최초 전체 FAQ 표시
    renderFaqs();
  </script>
